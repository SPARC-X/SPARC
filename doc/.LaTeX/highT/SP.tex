%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[allowframebreaks]{\texttt{SP\_FLAG}} \label{SP_FLAG}
\vspace*{-12pt}
\begin{columns}
\column{0.4\linewidth}
\begin{block}{Type}
Integer
\end{block}

\begin{block}{Default}
0
\end{block}

\column{0.4\linewidth}
\begin{block}{Unit}
No unit
\end{block}

\begin{block}{Example}
\texttt{SP\_FLAG}: 1
\end{block}
\end{columns}

\begin{block}{Description}
Flag to turn on SP method.
\end{block}

\begin{block}{Remark}
SP method cannot be turned on simultaneously with SQ, CS, SQ3, hybrid functionals. 
\end{block}

\end{frame}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[allowframebreaks]{\texttt{SP\_SMEARING}} \label{SP_SMEARING}
\vspace*{-12pt}
\begin{columns}
\column{0.4\linewidth}
\begin{block}{Type}
Double
\end{block}

\begin{block}{Default}
0.01
\end{block}

\column{0.4\linewidth}
\begin{block}{Unit}
Ha
\end{block}

\begin{block}{Example}
\texttt{SP\_SMEARING}: 0.05
\end{block}
\end{columns}

\begin{block}{Description}
Regularization parameter to ensure numerical stability of SPDFT method.
\end{block}

\begin{block}{Remark}
To avoid its effect on the physics of the problem, it is kept approx. 50x smaller than the Fermi-Dirac smearing.
\end{block}

\end{frame}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[allowframebreaks]{\texttt{{SP\_TOL\_OCC}}} \label{SP_TOL_OCC}
\vspace*{-12pt}
\begin{columns}
\column{0.4\linewidth}
\begin{block}{Type}
Double
\end{block}

\begin{block}{Default}
1e-12
\end{block}

\column{0.4\linewidth}
\begin{block}{Unit}
No unit
\end{block}

\begin{block}{Example}
\texttt{SP\_TOL\_OCC}: 1e-13
\end{block}
\end{columns}

\begin{block}{Description}
Allows obtaining the highest-energy plane wave in the spdft method, such that its occupation is less than this value.
\end{block}

\end{frame}

\begin{frame}[allowframebreaks]{\texttt{SP\_ISESPLIT\_CONST}} \label{SP_ISESPLIT_CONST}
\vspace*{-12pt}
\begin{columns}
\column{0.4\linewidth}
\begin{block}{Type}
Integer
\end{block}

\begin{block}{Default}
0
\end{block}

\column{0.4\linewidth}
\begin{block}{Unit}
No unit
\end{block}

\begin{block}{Example}
\texttt{SP\_ISESPLIT\_CONST}: 1
\end{block}
\end{columns}

\begin{block}{Description}
Flag to specify if the splitting energies in spdft method need to be kept constant while performing self-consistency. The constant value is read from .spDFT file. 
\end{block}

\end{frame}
